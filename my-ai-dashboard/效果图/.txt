"use client";

import { useEffect, useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import { BellRing, Eye, EyeOff, ArrowRight } from "lucide-react";

type RoleType = "管理员" | "门店经理" | "运营人员";

const roleOptions: RoleType[] = ["管理员", "门店经理", "运营人员"];

export default function AuthLoginCard() {
  const router = useRouter();

  const [role, setRole] = useState<RoleType>("管理员");
  const [account, setAccount] = useState("admin");
  const [password, setPassword] = useState("123456");
  const [remember, setRemember] = useState(true);
  const [showPassword, setShowPassword] = useState(false);
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState<{ type: "success" | "error"; text: string } | null>(null);

  useEffect(() => {
    const token = window.localStorage.getItem("restaurant_demo_token");
    if (token) {
      router.replace("/overview");
    }
  }, [router]);

  const greeting = useMemo(() => {
    const hour = new Date().getHours();
    if (hour < 6) return "凌晨值守中";
    if (hour < 12) return "早安，开始今日运营";
    if (hour < 18) return "下午好，查看午晚市状态";
    return "晚上好，收口今日经营数据";
  }, []);

  const fillDemo = (type: "admin" | "manager") => {
    if (type === "admin") {
      setRole("管理员");
      setAccount("admin");
      setPassword("123456");
    } else {
      setRole("门店经理");
      setAccount("manager01");
      setPassword("123456");
    }

    setMessage({
      type: "success",
      text: `已切换为${type === "admin" ? "管理员" : "门店经理"}演示账号。`,
    });
  };

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    setMessage(null);

    const username = account.trim();
    const pwd = password.trim();

    if (!username) {
      setMessage({ type: "error", text: "请输入账号后再登录。" });
      return;
    }

    if (!pwd) {
      setMessage({ type: "error", text: "请输入密码后再登录。" });
      return;
    }

    setLoading(true);

    try {
      await new Promise((resolve) => setTimeout(resolve, 900));

      window.localStorage.setItem("restaurant_demo_token", "demo-token");
      window.localStorage.setItem(
        "restaurant_demo_user",
        JSON.stringify({
          id: 1,
          username,
          display_name: role === "管理员" ? "系统管理员" : role,
          role,
        }),
      );
      window.localStorage.setItem("restaurant_selected_store_id", "1");

      setMessage({
        type: "success",
        text: `登录成功，正在进入后台。当前角色：${role}`,
      });

      router.push("/overview");
    } catch {
      setMessage({
        type: "error",
        text: "登录失败，请稍后重试。",
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <section className="w-full rounded-[30px] border border-white/70 bg-white/75 p-5 shadow-[0_10px_30px_rgba(48,37,28,0.08)] backdrop-blur md:p-7">
      <div className="flex flex-col gap-4 md:flex-row md:items-start md:justify-between">
        <div>
          <h2 className="text-3xl font-extrabold tracking-[-0.03em] text-stone-900 md:text-4xl">
            欢迎登录
          </h2>
          <p className="mt-2 text-sm text-stone-500">
            {greeting}，请输入账号信息进入 RestaurantOS 智能运营中心
          </p>
        </div>

        <div className="inline-flex w-fit items-center gap-2 rounded-2xl border border-orange-100 bg-orange-50 px-3 py-2 text-xs text-orange-700">
          <BellRing className="h-4 w-4" />
          当前有 3 条运营提醒
        </div>
      </div>

      <div className="mt-6 grid grid-cols-1 gap-3 sm:grid-cols-3">
        {roleOptions.map((item) => (
          <button
            key={item}
            type="button"
            onClick={() => {
              setRole(item);
              setMessage({
                type: "success",
                text: `当前登录角色已切换为：${item}`,
              });
            }}
            className={[
              "rounded-2xl border px-4 py-3 text-sm font-semibold transition",
              role === item
                ? "border-orange-200 bg-[linear-gradient(135deg,rgba(255,107,61,0.10),rgba(255,107,61,0.03))] text-orange-700"
                : "border-stone-200 bg-white text-stone-600 hover:-translate-y-0.5 hover:shadow-sm",
            ].join(" ")}
          >
            {item}
          </button>
        ))}
      </div>

      <form onSubmit={handleSubmit} className="mt-6 space-y-5">
        <div className="space-y-2">
          <label className="text-sm font-semibold text-stone-700">账号</label>
          <input
            type="text"
            value={account}
            onChange={(e) => setAccount(e.target.value)}
            placeholder="请输入账号 / 用户名 / 邮箱"
            className="h-14 w-full rounded-2xl border border-stone-200 bg-white px-4 text-sm outline-none transition focus:border-orange-300 focus:ring-4 focus:ring-orange-100"
          />
        </div>

        <div className="space-y-2">
          <label className="text-sm font-semibold text-stone-700">密码</label>
          <div className="relative">
            <input
              type={showPassword ? "text" : "password"}
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="请输入登录密码"
              className="h-14 w-full rounded-2xl border border-stone-200 bg-white px-4 pr-14 text-sm outline-none transition focus:border-orange-300 focus:ring-4 focus:ring-orange-100"
            />
            <button
              type="button"
              onClick={() => setShowPassword((v) => !v)}
              className="absolute right-3 top-1/2 grid h-9 w-9 -translate-y-1/2 place-items-center rounded-xl bg-stone-100 text-stone-500 transition hover:bg-stone-200 hover:text-stone-700"
              aria-label="切换密码显示"
            >
              {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
            </button>
          </div>
        </div>

        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
          <label className="inline-flex items-center gap-2 text-sm text-stone-600">
            <input
              type="checkbox"
              checked={remember}
              onChange={(e) => setRemember(e.target.checked)}
              className="h-4 w-4 accent-orange-500"
            />
            <span>记住我</span>
          </label>

          <button
            type="button"
            onClick={() =>
              setMessage({
                type: "success",
                text: "这里后续可以接入忘记密码流程，例如短信或邮箱找回。",
              })
            }
            className="text-sm font-semibold text-orange-600 hover:text-orange-700"
          >
            忘记密码？
          </button>
        </div>

        <button
          type="submit"
          disabled={loading}
          className="flex h-14 w-full items-center justify-center gap-2 rounded-2xl bg-[linear-gradient(135deg,#ff6b3d,#ff8a5b)] text-sm font-bold text-white shadow-[0_10px_22px_rgba(255,107,61,0.25)] transition hover:-translate-y-0.5 disabled:cursor-not-allowed disabled:opacity-70"
        >
          <span>{loading ? "登录中..." : "进入 RestaurantOS"}</span>
          {!loading && <ArrowRight className="h-4 w-4" />}
        </button>

        {message && (
          <div
            className={[
              "rounded-2xl border px-4 py-3 text-sm leading-7",
              message.type === "success"
                ? "border-emerald-200 bg-emerald-50 text-emerald-700"
                : "border-red-200 bg-red-50 text-red-700",
            ].join(" ")}
          >
            {message.text}
          </div>
        )}
      </form>

      <div className="my-6 flex items-center gap-3">
        <div className="h-px flex-1 bg-stone-200" />
        <span className="text-xs text-stone-400">快捷入口</span>
        <div className="h-px flex-1 bg-stone-200" />
      </div>

      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
        <button
          type="button"
          onClick={() => fillDemo("admin")}

  children,
}: {

}) {

  const [ready, setReady] = useState(false);
  const [displayName, setDisplayName] = useState("系统管理员");

  useEffect(() => {
    const token = window.localStorage.getItem("restaurant_demo_token");
    const rawUser = window.localStorage.getItem("restaurant_demo_user");

    if (!token) {
      router.replace("/login");
      return;
    }

    if (rawUser) {
      try {
        const parsed = JSON.parse(rawUser) as { display_name?: string };
        if (parsed.display_name) {
          setDisplayName(parsed.display_name);
        }
      } catch {
        // ignore
      }
    }

    setReady(true);
  }, [router]);

  const currentTime = useMemo(() => {
    const now = new Date();
    return now.toLocaleString("zh-CN", {
      month: "2-digit",
      day: "2-digit",
      weekday: "short",
      hour: "2-digit",
      minute: "2-digit",
    });
  }, []);

  const handleLogout = () => {
    window.localStorage.removeItem("restaurant_demo_token");
    window.localStorage.removeItem("restaurant_demo_user");
    window.localStorage.removeItem("restaurant_selected_store_id");
    router.replace("/login");
  };

  if (!ready) {
    return (
      <div className="min-h-screen bg-[#f6f3f0] p-4">

            <div className="grid h-11 w-11 place-items-center rounded-2xl bg-[linear-gradient(135deg,#ff6b3d,#ff9a62)] text-white shadow-[0_8px_18px_rgba(255,107,61,0.28)]">
              <ChefHat className="h-5 w-5" />

              <div className="text-xl font-extrabold tracking-[-0.03em] text-stone-900">

              <div className="text-xs text-stone-500">智能运营中心</div>
